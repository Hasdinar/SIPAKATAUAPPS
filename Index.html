          const matchSearch = s.full_name.toLowerCase().includes(searchQuery.toLowerCase()) || s.nis.includes(searchQuery);
          return matchClass && matchSearch;
        });

      const handleFileUpload = (e) => {
        const file = e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = (evt) => {
          try {
            const data = new Uint8Array(evt.target.result);
            const workbook = XLSX.read(data, { type: 'array' });
            const sheetName = workbook.SheetNames[0];
            const sheet = workbook.Sheets[sheetName];
            const json = XLSX.utils.sheet_to_json(sheet);

            // Validate
            const valid = [];
            const errors = [];

            json.forEach((row, idx) => {
              const rowNo = idx + 2;
              const name = row['Nama Lengkap'] || row['Nama'] || '';
              const nis = String(row['NIS'] || '').trim();
              const gender = String(row['Jenis Kelamin'] || 'L').toUpperCase().startsWith('P') ? 'P' : 'L';

              if (!name || !nis) {
                errors.push({ row: rowNo, reason: 'Nama Lengkap atau NIS kosong' });
                return;

              const isDup = students.some(s => s.class_id === filterClass && s.nis === nis);

                errors.push({ row: rowNo, reason: `NIS ${nis} sudah ada di database` });
                return;

                id: 's_imp_' + Date.now() + '_' + idx,
                class_id: filterClass,
                full_name: name,
                nis: nis,
                nisn: row['NISN'] ? String(row['NISN']) : '',
                gender: gender,

                dob: row['Tanggal Lahir'] || '',
                address: row['Alamat'] || ''

    function AttendanceView({ session, classes, students, attendanceSessions, attendanceRecords, onSaveSession, onDeleteSession }) {
      const isTeacher = session.role === 'teacher';
      const [selectedClass, setSelectedClass] = useState(classes[0]?.id || '');
      const [activeSessionId, setActiveSessionId] = useState('');
      
      // Form states for recording
      const [meetingNo, setMeetingNo] = useState(attendanceSessions.length + 1);
      const [meetingDate, setMeetingDate] = useState(new Date().toISOString().split('T')[0]);
      const [topic, setTopic] = useState('');
      const [tempStatuses, setTempStatuses] = useState({});
      const [tempNotes, setTempNotes] = useState({});

      const classStudents = useMemo(() => students.filter(s => s.class_id === selectedClass), [students, selectedClass]);
      const classSessions = useMemo(() => attendanceSessions.filter(s => s.class_id === selectedClass), [attendanceSessions, selectedClass]);

      // Load session details into form
      useEffect(() => {
        if (activeSessionId) {
          const sess = attendanceSessions.find(s => s.id === activeSessionId);
          if (sess) {
            setMeetingNo(sess.meeting_no);
            setMeetingDate(sess.date);
            setTopic(sess.topic);

            const mapS = {}, mapN = {};
            attendanceRecords.filter(r => r.session_id === activeSessionId).forEach(r => {
              mapS[r.student_id] = r.status;
              mapN[r.student_id] = r.notes || '';
            });
            setTempStatuses(mapS);
            setTempNotes(mapN);
          }
        } else {
          // Reset for new session
          setMeetingNo(classSessions.length + 1);
          setTopic('');
          const mapS = {}, mapN = {};
          classStudents.forEach(s => { mapS[s.id] = 'Hadir'; mapN[s.id] = ''; });
          setTempStatuses(mapS);
          setTempNotes(mapN);
        }
      }, [activeSessionId, selectedClass, attendanceSessions, attendanceRecords, classStudents]);

      const handleSave = (e) => {
        e.preventDefault();
        const sessId = activeSessionId || 'att_' + Date.now();
        const sessObj = {
          id: sessId,
          class_id: selectedClass,
          meeting_no: parseInt(meetingNo),
          date: meetingDate,
          topic: topic || `Pertemuan ${meetingNo}`
        };

        const recordObjs = classStudents.map(st => ({
          id: 'ar_' + sessId + '_' + st.id,
          session_id: sessId,
          student_id: st.id,
          status: tempStatuses[st.id] || 'Hadir',
          notes: tempNotes[st.id] || ''
        }));

        alert('Data kehadiran pertemuan berhasil disimpan!');

                  {classes.map(c => <option key={c.id} value={c.id}>Kelas {c.name}</option>)}
                </select>
              </div>

                <label className="block text-[10px] font-bold text-slate-400 uppercase tracking-wider mb-1">Riwayat Sesi Pertemuan</label>

    function AssessmentSOLOView({ session, classes, students, curriculum, soloScores, onSaveScores }) {
      const isTeacher = session.role === 'teacher';
      const [selectedClass, setSelectedClass] = useState(classes[0]?.id || '');
      
      // Select Curriculum TP
      const activeCp = curriculum[0];
      const activeMat = activeCp?.materials[0];
      const activeTp = activeMat?.tps[0];

      const classStudents = useMemo(() => students.filter(s => s.class_id === selectedClass), [students, selectedClass]);
      const [localScores, setLocalScores] = useState({ ...soloScores });

      const handleScoreChange = (tpId, ikId, studentId, level) => {
        if (!isTeacher) return;
        const key = `${tpId}_${ikId}_${studentId}`;
        setLocalScores(prev => ({ ...prev, [key]: parseInt(level) }));
      };

      const handleSave = () => {
        onSaveScores(localScores);
        alert('Penilaian Taksonomi SOLO dan konversi angka berhasil disimpan!');
      };

      // Exact Deterministic Conversion Formula Calculation
      // Formula: Score = MinRange + (AvgSolo - BaseLevel) * IntervalWidth
      const calculateConversion = (studentId) => {
        if (!activeTp) return { avgSolo: 0, levelName: 'Belum Dinilai', convertedScore: 0, status: 'Belum Lengkap' };

        const iktps = activeTp.iktps;
        let sum = 0, count = 0;

        iktps.forEach(ik => {
          const score = localScores[`${activeTp.id}_${ik.id}_${studentId}`];
          if (score) {
            sum += score;
            count++;
          }
        });

        if (count < iktps.length) {
          return { avgSolo: 0, levelName: 'Belum Lengkap', convertedScore: 0, status: 'Belum Lengkap' };
        }

        const avgSolo = sum / iktps.length; // e.g. 11/3 = 3.67
        const config = activeTp.kktp_config;

        let levelName = 'Prestructural';
        let baseMin = config.prestructural_min;
        let baseLevel = 1;

        if (avgSolo >= 4.5) {
          levelName = 'Extended Abstract';
          baseMin = config.extended_min;
          baseLevel = 4;
        } else if (avgSolo >= 3.5) {
          levelName = 'Relational';
          baseMin = config.relational_min; // 84 or 75 config
          baseLevel = 3;
        } else if (avgSolo >= 2.5) {
          levelName = 'Multistructural';
          baseMin = config.multistructural_min; // 75
          baseLevel = 2;
        } else if (avgSolo >= 1.5) {
          levelName = 'Unistructural';
          baseMin = config.unistructural_min; // 60
          baseLevel = 1;
        }

        // Example match: (3.67 - 3) * 8 + 75 = 80.36
        const converted = baseMin + (avgSolo - baseLevel) * config.interval_width;
        const finalScore = Math.min(100, Math.max(0, parseFloat(converted.toFixed(2))));

        // KKTP Status
        const kktpMinLevel = activeTp.kktp_level;
        const levelOrder = ['Prestructural', 'Unistructural', 'Multistructural', 'Relational', 'Extended Abstract'];
        const isAchieved = levelOrder.indexOf(levelName) >= levelOrder.indexOf(kktpMinLevel);

        return {
          avgSolo: avgSolo.toFixed(2),
          levelName: levelName,
          convertedScore: finalScore,
          status: isAchieved ? 'Tercapai' : 'Belum Tercapai'
        };
      };

      return (
        <div className="space-y-6 max-w-6xl mx-auto">
          {/* Header Controls */}
          <div className="bg-white p-5 rounded-2xl border border-slate-200/80 shadow-sm flex flex-col md:flex-row md:items-center justify-between gap-4">
            <div>
              <label className="block text-[10px] font-bold text-slate-400 uppercase tracking-wider mb-1">Pilih Kelas</label>
              <select value={selectedClass} onChange={e => setSelectedClass(e.target.value)} className="px-3.5 py-2 bg-slate-50 border border-slate-200 rounded-xl text-xs font-bold text-slate-800 focus:ring-2 focus:ring-primary-500 outline-none">
                {classes.map(c => <option key={c.id} value={c.id}>Kelas {c.name}</option>)}
              </select>
            </div>

            <div className="p-3 bg-primary-50 border border-primary-100 rounded-xl text-xs text-primary-800 flex items-center space-x-2">
              <i data-lucide="calculator" className="w-4 h-4 text-primary-600 flex-shrink-0"></i>
              <span>Rumus Konversi: <b>Nilai = Batas Bawah + (Rata SOLO â Base Level) Ã Interval</b></span>
            </div>
          </div>

          {/* Active TP Formulation Summary */}
          {activeTp && (
            <div className="bg-white p-5 rounded-2xl border border-slate-200/80 shadow-sm space-y-2">
              <div className="flex justify-between items-center">
                <span className="text-xs font-mono font-bold text-primary-700 bg-primary-50 px-2.5 py-0.5 rounded">{activeTp.code}</span>
                <span className="text-xs font-bold text-emerald-700">Target KKTP: {activeTp.kktp_level}</span>
              </div>
              <h4 className="font-bold text-slate-800 text-sm">{activeTp.formulation}</h4>
            </div>
          )}

          {/* Assessment Table */}

              {isTeacher && (
                <button onClick={handleSave} className="px-4 py-2 bg-primary-600 hover:bg-primary-700 text-white font-bold text-xs rounded-xl shadow-md transition flex items-center space-x-1.5">
                  <i data-lucide="save" className="w-3.5 h-3.5"></i>
                  <span>Simpan Penilaian</span>
                </button>
              )}
            </div>

            <div className="overflow-x-auto">
              <table className="w-full text-left border-collapse text-xs">
                <thead>
                  <tr className="bg-slate-50 border-b border-slate-200 text-slate-500 font-bold uppercase tracking-wider">
                    <th className="py-3 px-4">Nama Murid</th>
                    {activeTp?.iktps.map((ik, i) => (
                      <th key={ik.id} className="py-3 px-3 text-center max-w-xs">
                        IKTP {i + 1}
                      </th>
                    ))}
                    <th className="py-3 px-3 text-center bg-purple-50/50">Rata SOLO</th>
                    <th className="py-3 px-3 text-center bg-purple-50/50">Level SOLO</th>
                    <th className="py-3 px-3 text-center bg-purple-50/50">Nilai Konversi</th>
                    <th className="py-3 px-3 text-center">Status KKTP</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 font-medium">
                  {classStudents.map(st => {
                    const calc = calculateConversion(st.id);
                    return (
                      <tr key={st.id} className="hover:bg-slate-50/80 transition">
                        <td className="py-3 px-4 font-bold text-slate-800">{st.full_name}</td>
                        
                        {activeTp?.iktps.map(ik => {
                          const val = localScores[`${activeTp.id}_${ik.id}_${st.id}`] || '';
                          return (

                              <select 
                                disabled={!isTeacher}

                                onChange={e => handleScoreChange(activeTp.id, ik.id, st.id, e.target.value)}
                                className={`px-2 py-1 border rounded-lg text-xs font-bold outline-none ${
                                  val == 1 ? 'bg-red-50 text-red-600 border-red-200' :
                                  val == 2 ? 'bg-orange-50 text-orange-600 border-orange-200' :

                                  val == 5 ? 'bg-purple-50 text-purple-600 border-purple-200' : 'bg-slate-50 text-slate-400'
                                }`}
                              >
                                <option value="">- Level -</option>
                                <option value="1">1. Prestructural</option>
                                <option value="2">2. Unistructural</option>
                                <option value="3">3. Multistructural</option>
                                <option value="4">4. Relational</option>
                                <option value="5">5. Extended Abstract</option>
                              </select>
                            </td>
                          );
                        })}

                        <td className="py-3 px-3 text-center font-bold font-mono bg-purple-50/30 text-purple-900">{calc.avgSolo}</td>
                        <td className="py-3 px-3 text-center font-bold bg-purple-50/30 text-purple-900">{calc.levelName}</td>

                          <span className={`px-2.5 py-1 rounded-full text-[10px] font-bold ${
